## 2. Smoke Tests (Jest Unit Tests)

These are the Jest-based tests that must pass when `npm run test` is executed.
The default NestJS scaffold already generates `src/app.controller.spec.ts`. The Dev agent
must ensure this test file passes. Below is the expected test specification.

### 2.1 Default App Controller Test (`src/app.controller.spec.ts`)

```typescript
import { Test, TestingModule } from '@nestjs/testing';
import { AppController } from './app.controller';
import { AppService } from './app.service';

describe('AppController', () => {
  let appController: AppController;

  beforeEach(async () => {
    const app: TestingModule = await Test.createTestingModule({
      controllers: [AppController],
      providers: [AppService],
    }).compile();

    appController = app.get<AppController>(AppController);
  });

  describe('root', () => {
    it('should be defined', () => {
      expect(appController).toBeDefined();
    });

    it('should return "Hello World!" when getHello is called', () => {
      expect(appController.getHello()).toBe('Hello World!');
    });
  });
});
```

**Test IDs:**

| ID | Test Description | Pattern |
|----|-----------------|---------|
| ST-01 | App module compiles successfully | `TestingModule` compiles without error |
| ST-02 | AppController is defined | `expect(appController).toBeDefined()` |
| ST-03 | getHello returns expected string | `expect(appController.getHello()).toBe('Hello World!')` |

### 2.2 E2E Smoke Test (`test/app.e2e-spec.ts`)

The default NestJS scaffold includes an e2e test. It must pass with `npm run test:e2e`.

| ID | Test Description | Expected |
|----|-----------------|----------|
| ST-04 | `GET /` returns 200 | HTTP status 200 |
| ST-05 | `GET /` returns "Hello World!" | Response body = "Hello World!" |

## 4. Combined Validation Script

A single script the Dev agent can run to validate all checks at once.

```bash
#!/bin/bash
# S-01.1 Full Validation Script
# Run from weatherflow/ root
set -e
PASS=0
FAIL=0

check() {
  local desc="$1"
  shift
  if eval "$@" > /dev/null 2>&1; then
    echo "  PASS: $desc"
    ((PASS++))
  else
    echo "  FAIL: $desc"
    ((FAIL++))
  fi
}

echo "=== Section 1: Runtime & CLI ==="
check "Node >= 20" "node -e \"process.exit(parseInt(process.version.slice(1)) >= 20 ? 0 : 1)\""
check "NestJS CLI v11" "nest --version | grep -q '^11'"

echo ""
echo "=== Section 2: tsconfig.json ==="
check "strict: true" "node -e \"const t=require('./tsconfig.json'); process.exit(t.compilerOptions.strict===true?0:1)\""
check "noUnusedLocals: true" "node -e \"const t=require('./tsconfig.json'); process.exit(t.compilerOptions.noUnusedLocals===true?0:1)\""
check "noUnusedParameters: true" "node -e \"const t=require('./tsconfig.json'); process.exit(t.compilerOptions.noUnusedParameters===true?0:1)\""

echo ""
echo "=== Section 3: package.json scripts ==="
for s in "start:dev" "build" "test" "lint" "test:cov" "format" "start" "start:prod" "test:e2e"; do
  check "script: $s" "node -e \"const p=require('./package.json'); process.exit(p.scripts['$s']?0:1)\""
done

echo ""
echo "=== Section 4: Production dependencies ==="
for d in "@nestjs/mongoose" "mongoose" "@nestjs/swagger" "@nestjs/config" "dotenv" \
         "class-validator" "class-transformer" "axios" "@nestjs/jwt" "@nestjs/passport" \
         "passport" "passport-jwt" "bcrypt" "@nestjs/event-emitter"; do
  check "dep: $d" "node -e \"const p=require('./package.json'); process.exit(p.dependencies['$d']?0:1)\""
done

echo ""
echo "=== Section 5: Dev dependencies ==="
for d in "@types/passport-jwt" "@types/bcrypt" "@swc/core" "@swc/jest"; do
  check "devDep: $d" "node -e \"const p=require('./package.json'); process.exit(p.devDependencies['$d']?0:1)\""
done

echo ""
echo "=== Section 6: .env.example ==="
check ".env.example exists" "test -f .env.example"
for v in PORT MONGODB_URI JWT_SECRET JWT_EXPIRES_IN TELEGRAM_BOT_TOKEN; do
  check ".env.example has $v" "grep -q '^${v}=' .env.example"
done

echo ""
echo "=== Section 7: .gitignore ==="
check ".gitignore has node_modules" "grep -q 'node_modules' .gitignore"
check ".gitignore has dist" "grep -q 'dist' .gitignore"
check ".gitignore has .env" "grep -qE '^\\.env$' .gitignore"
check ".gitignore has coverage" "grep -q 'coverage' .gitignore"

echo ""
echo "=== Section 8: Project structure ==="
for f in src/main.ts src/app.module.ts src/app.controller.ts src/app.service.ts \
         src/app.controller.spec.ts test/app.e2e-spec.ts nest-cli.json .prettierrc; do
  check "file: $f" "test -f $f"
done
check "ESLint config" "test -f .eslintrc.js || test -f eslint.config.mjs"

echo ""
echo "=== Section 9: Build & Lint ==="
check "npm run build" "npm run build"
check "npm run lint" "npm run lint"
check "npm run test" "npm run test"
check "npm run format" "npm run format"

echo ""
echo "=============================="
echo "PASSED: $PASS"
echo "FAILED: $FAIL"
echo "=============================="
[ "$FAIL" -eq 0 ] && echo "ALL CHECKS PASSED" || echo "SOME CHECKS FAILED"
exit $FAIL
```
